% ─── RESEARCH & PUBLICATIONS ─────────────────────────────────────────────────
\resumeSection{Research \& Publications}

\textbf{\href{https://doi.org/10.1007/978-3-032-10202-7_33}
{A Benchmark Dataset and Evaluation Framework for Vietnamese Large Language Models in Customer Support}}

\textit{Co-author} --- 17th International Conference on Computational Collective Intelligence (ICCCI 2025)
\hfill 2025

\begin{itemize}
    \item \textbf{Dataset:} Introduced CSConDa, a benchmark comprising 9,849 anonymized QA pairs from real-world Vietnamese customer--advisor conversations, categorized into General, Domain-Specific Simple, and Domain-Specific Complex queries.

    \item \textbf{Evaluation Framework:} Proposed a multi-aspect evaluation framework combining six automatic metrics (lexical, semantic, and hallucination assessment) with five syntactic measures to evaluate accuracy, fluency, and linguistic characteristics of Vietnamese LLMs.

    \item \textbf{Empirical Findings:} Benchmarked 11 lightweight open-source ViLLMs (7--9B parameters), identifying Vistral 7B as the top-ranked model and revealing that LLMs tend to generate more verbose and syntactically rigid responses than human advisors.
\end{itemize}

Published in \textit{Advances in Computational Collective Intelligence},
Springer CCIS, vol.~2747, pp.~481--496.
\href{https://arxiv.org/abs/2507.22542}{arXiv:2507.22542}.
